    end

local guiBusy = false

local function minimizeGui()
    if guiBusy then return end
    guiBusy = true
    Main.Visible = false
    MinBar.Position = Main.Position
    MinBar.Visible = true
    guiBusy = false
end

local function restoreGui()
    if guiBusy then return end
    guiBusy = true
    MinBar.Visible = false
    Main.Position = MinBar.Position
    Main.Visible = true
    guiBusy = false
end

local function createBaseplate()
    if not safePart or not safePart.Parent then
        safePart = Instance.new("Part")
        safePart.Size = Vector3.new(25, 1, 25)
        safePart.Position = Vector3.new(0, 600, 0)
        safePart.Anchored = true
        safePart.Transparency = 0.4
        safePart.BrickColor = BrickColor.new("Pastel yellow")
        safePart.Name = "SafeBaseplate"
        safePart.Parent = Workspace
    end
end

local function removeBaseplate()
    if safePart then
        safePart:Destroy()
        safePart = nil
    end
end

local function normalTeleport(targetCFrame)
    local char = player.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        char.HumanoidRootPart.CFrame = targetCFrame
    end
end

local function weirdJumpTeleport(targetCFrame)
    local char = player.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        local hrp = char.HumanoidRootPart
        local hum = char:FindFirstChildOfClass("Humanoid")
        
        local randomAngle = math.rad(math.random(0, 360))
        hrp.CFrame = targetCFrame * CFrame.Angles(0, randomAngle, math.rad(math.random(-20, 20)))
        
        if hum then
            for i = 1, 3 do
                hum.Jump = true
                hrp.Velocity = Vector3.new(math.random(-8, 8), math.random(10, 18), math.random(-8, 8))
                task.wait(0.2)
            end
        end
        
        hrp.CFrame = targetCFrame
    end
end

local function getCoinContainer()
    if Workspace:FindFirstChild("CoinContainer") then
        return Workspace.CoinContainer
    end
    for _, child in ipairs(Workspace:GetChildren()) do
        if child:FindFirstChild("CoinContainer") then
            return child.CoinContainer
        end
    end
    return nil
end

local function isRoundActive()
    local coinContainer = getCoinContainer()
    if coinContainer and #coinContainer:GetChildren() > 0 then
        return true
    end
    return false
end

-- Bucle de recolección optimizado
task.spawn(function()
    while true do
        task.wait(0.6)
        
        if enabled then
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            
            if not isRoundActive() or finishedCurrentRound then
                if not isRoundActive() then
                    coinsCollected = 0
                    finishedCurrentRound = false
                end
                removeBaseplate()
                doAntiAFKMovement()
                task.wait(2.5)
                continue
            end

            if not char or not hum or hum.Health <= 0 then
                task.wait(1.0)
                continue
            end

            createBaseplate()
            local safeCFrame = CFrame.new(safePart.Position + Vector3.new(0, 3, 0))
            
            normalTeleport(safeCFrame)
            task.wait(1.2)

            local coinContainer = getCoinContainer()
            
            if coinContainer then
                for _, coin in ipairs(coinContainer:GetChildren()) do
                    if not enabled or not isRoundActive() or hum.Health <= 0 or finishedCurrentRound then
                        break
                    end

                    local coinPart = coin:IsA("BasePart") and coin or coin:FindFirstChild("CoinVisual") or coin:FindFirstChild("Hitbox")
                    
                    if coinPart and coinPart:IsA("BasePart") and coinPart.Parent then
                        
                        -- Si ya supera o llega a 53 monedas, simplemente se queda en el mapa y termina la ronda para el script
                        if coinsCollected >= 53 then
                            finishedCurrentRound = true
                            removeBaseplate()
                            break
                        end

                        weirdJumpTeleport(coinPart.CFrame + Vector3.new(0, 1, 0))
                        
                        task.wait(0.5)
                        coinsCollected = coinsCollected + 1
                        
                        if coinsCollected >= 53 then
                            finishedCurrentRound = true
                            
                            if coin and coin.Parent then
                                coin:Destroy()
                            end
                            
                            removeBaseplate()
                            break
                        end
                        
                        if not finishedCurrentRound and enabled then
                            normalTeleport(safeCFrame)
                            task.wait(1.8)
                        end
                    end
                end
            end
        end
    end
end)

local dragging, dragStart, startPos = false, nil, nil
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
Header.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local mDragging, mDragStart, mStartPos = false, nil, nil
MinBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        mDragging = true
        mDragStart = input.Position
        mStartPos = MinBar.Position
    end
end)
MinBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if mDragging and mDragStart then
            local delta = input.Position - mDragStart
            if delta.Magnitude < 8 then
                task.spawn(restoreGui)
            end
        end
        mDragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
    if mDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - mDragStart
        MinBar.Position = UDim2.new(mStartPos.X.Scale, mStartPos.X.Offset + delta.X, mStartPos.Y.Scale, mStartPos.Y.Offset + delta.Y)
    end
end)
